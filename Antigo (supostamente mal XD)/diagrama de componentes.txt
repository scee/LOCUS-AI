' =====================================================
' 1. CLIENTE / APRESENTAÇÃO
' =====================================================

package "Cliente / Apresentação" as Cliente {
    component "InterfaceWeb\n(Dashboard LOCUS AI)" as InterfaceWeb
}

' =====================================================
' 2. LÓGICA DE NEGÓCIO E ANÁLISE
' =====================================================

package "Lógica de Negócio e Análise" as Negocio {
    interface "IAplicacao" as IAplicacao
    component "GestorAplicacao" as GestorAplicacao

    interface "IAnalise" as IAnalise
    component "MotorAnaliseTerritorial" as MotorAnalise

    interface "IML" as IML
    component "ModuloML" <<opcional>> as ModuloML

    ' Ancorar interfaces diretamente à esquerda/topo dos respetivos componentes
    IAplicacao -down- GestorAplicacao
    IAnalise -right- MotorAnalise
    IML -right- ModuloML
}

' =====================================================
' 3. INTEGRAÇÃO E DADOS
' =====================================================

package "Integração e Dados" as Dados {
    interface "IPipeline" as IPipeline
    component "PipelineETL" as PipelineETL

    interface "IMetadados" as IMetadados
    component "GestorMetadados" as GestorMetadados

    interface "IDados" as IDados
    database "RepositorioAnalitico\nPostgreSQL + PostGIS" as Repositorio

    ' Ancorar interfaces diretamente aos componentes de dados
    IPipeline -right- PipelineETL
    IMetadados -right- GestorMetadados
    IDados -right- Repositorio
}

' =====================================================
' 4. SISTEMAS EXTERNOS
' =====================================================

package "Sistemas Externos" as Externos {
    component "E-REDES_Data" as EREDES <<external>>
    component "Pordata" as Pordata <<external>>
}

' =====================================================
' 5. DEPENDÊNCIAS ENTRE COMPONENTES
' =====================================================

' Interface e camada de aplicação
InterfaceWeb ..> IAplicacao : utiliza

' Gestor da aplicação coordena as funcionalidades
GestorAplicacao ..> IAnalise : solicita análises
GestorAplicacao ..> IPipeline : solicita operações\nde dados
GestorAplicacao ..> IMetadados : consulta metadados
GestorAplicacao ..> IML : solicita ML\n[se disponível]

' Motor de análise
MotorAnalise ..> IDados : consulta dados
MotorAnalise ..> IML : utiliza resultados\n[condicional]

' Pipeline e Catálogo
PipelineETL ..> IDados : escreve dados\nbrutos e preparados
PipelineETL ..> IMetadados : regista proveniência
GestorMetadados ..> IDados : consulta / regista\nmetadados

' Fontes externas
PipelineETL ..> EREDES : adquire dados
PipelineETL ..> Pordata : adquire indicadores

' =====================================================
' 6. NOTAS ARQUITETURAIS
' =====================================================

note bottom of Repositorio
  <b>Separação lógica dos dados:</b>

end note

  - Normalização
  - Validação
end note
